cmake_minimum_required(VERSION 3.22)
project(LamixBootContract LANGUAGES C CXX)
get_filename_component(LAMIX_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED YES)
set(CMAKE_C_EXTENSIONS NO)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED YES)
set(CMAKE_CXX_EXTENSIONS NO)
option(LAMIX_SANITIZERS "ASan/UBSan host tests" OFF)
add_compile_options(-Wall -Wextra -Werror -Wpedantic)
include_directories("${LAMIX_ROOT}/include")
if(CMAKE_CROSSCOMPILING)
  add_library(boot-contract STATIC "${LAMIX_ROOT}/lib/boot/contract.c" "${LAMIX_ROOT}/lib/boot/memory.c" layout.c layout.cpp)
  target_compile_definitions(boot-contract PRIVATE LAMIX_COMPILE_ONLY=1)
  target_compile_options(boot-contract PRIVATE -ffreestanding -fno-builtin -fno-stack-protector -mno-red-zone -mgeneral-regs-only -fno-pie)
  target_compile_options(boot-contract PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-fno-exceptions;-fno-rtti>)
else()
  find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
  if(LAMIX_SANITIZERS)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
  endif()
  add_library(boot-contract SHARED "${LAMIX_ROOT}/lib/boot/contract.c" "${LAMIX_ROOT}/lib/boot/memory.c")
  add_executable(layout-report report.c)
  add_executable(layout-c layout.c)
  add_executable(layout-cpp layout.cpp)
  # Runner C kiểm tra mutations dưới ASan/UBSan.
  add_executable(contract-native native.c "${LAMIX_ROOT}/lib/boot/contract.c" "${LAMIX_ROOT}/lib/boot/memory.c")
  enable_testing()
  file(MAKE_DIRECTORY "${LAMIX_ROOT}/out/m03")
  add_test(NAME layout-c COMMAND layout-c)
  add_test(NAME layout-cpp COMMAND layout-cpp)
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/write_fixture.py" "${CMAKE_CURRENT_BINARY_DIR}/bootinfo.bin" COMMAND_ERROR_IS_FATAL ANY)
  add_test(NAME contract-native COMMAND contract-native "${CMAKE_CURRENT_BINARY_DIR}/bootinfo.bin")
  if(LAMIX_SANITIZERS)
    # Không heap trong thư viện; LSan không chạy được dưới ptrace sandbox.
    set_tests_properties(layout-c layout-cpp contract-native PROPERTIES ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0;UBSAN_OPTIONS=halt_on_error=1:print_stacktrace=1")
  endif()
  if(NOT LAMIX_SANITIZERS)
    add_test(NAME contract-python COMMAND "${Python3_EXECUTABLE}" -m pytest -q "${CMAKE_CURRENT_SOURCE_DIR}" "--junitxml=${LAMIX_ROOT}/out/m03/host-${CMAKE_BUILD_TYPE}.xml")
    set_tests_properties(contract-python PROPERTIES WORKING_DIRECTORY "${LAMIX_ROOT}" ENVIRONMENT "LAMIX_CONTRACT_LIBRARY=$<TARGET_FILE:boot-contract>" TIMEOUT 60)
  endif()
endif()
