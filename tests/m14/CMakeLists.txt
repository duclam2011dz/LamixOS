cmake_minimum_required(VERSION 3.22)
project(LamixM14 LANGUAGES C)
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED YES)
set(CMAKE_C_EXTENSIONS NO)
add_compile_options(-Wall -Wextra -Werror -Wpedantic -fsanitize=address,undefined -fno-omit-frame-pointer)
add_link_options(-fsanitize=address,undefined)
include_directories("${CMAKE_CURRENT_SOURCE_DIR}/../../kernel" "${CMAKE_CURRENT_SOURCE_DIR}/../../include")
enable_testing()
file(MAKE_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../out/m14")
add_executable(m14-native native.c "${CMAKE_CURRENT_SOURCE_DIR}/../../kernel/acpi.c" "${CMAKE_CURRENT_SOURCE_DIR}/../../kernel/apic.c")
add_test(NAME m14-native-sanitize COMMAND m14-native)
set_tests_properties(m14-native-sanitize PROPERTIES TIMEOUT 180 ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0;UBSAN_OPTIONS=halt_on_error=1:print_stacktrace=1")
add_test(NAME m14-host COMMAND "${Python3_EXECUTABLE}" -m pytest -q "${CMAKE_CURRENT_SOURCE_DIR}/test_m14.py" -m "not qemu" "--junitxml=${CMAKE_CURRENT_SOURCE_DIR}/../../out/m14/host.xml")
set_tests_properties(m14-host PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../.." TIMEOUT 300)
add_test(NAME m14-bios-uefi-guest COMMAND "${Python3_EXECUTABLE}" -m pytest -q "${CMAKE_CURRENT_SOURCE_DIR}/test_m14.py" -m qemu "--junitxml=${CMAKE_CURRENT_SOURCE_DIR}/../../out/m14/guest.xml")
set_tests_properties(m14-bios-uefi-guest PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../.." TIMEOUT 1800 ENVIRONMENT "LAMIX_STAGE2_BUILD=${CMAKE_CURRENT_SOURCE_DIR}/../../build/m14-bios;LAMIX_UEFI_BUILD=${CMAKE_CURRENT_SOURCE_DIR}/../../build/m14-uefi")
