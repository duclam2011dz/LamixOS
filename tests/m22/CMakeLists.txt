cmake_minimum_required(VERSION 3.22)
project(LamixM22 C)
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED YES)
set(CMAKE_C_EXTENSIONS NO)
add_compile_options(-Wall -Wextra -Werror -Wpedantic)
include_directories("${CMAKE_CURRENT_SOURCE_DIR}/../../kernel")
enable_testing()
get_filename_component(LAMIX_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
file(MAKE_DIRECTORY "${LAMIX_ROOT}/out/m22")
add_executable(m22-native native.c ../../kernel/handle.c ../../kernel/ipc.c)
target_compile_options(m22-native PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
target_link_options(m22-native PRIVATE -fsanitize=address,undefined)
add_test(NAME m22-native-sanitize COMMAND m22-native)
set_tests_properties(m22-native-sanitize PROPERTIES TIMEOUT 180 ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0;UBSAN_OPTIONS=halt_on_error=1:print_stacktrace=1")
add_executable(m22-concurrency-tsan concurrency.c ../../kernel/handle.c ../../kernel/ipc.c)
target_compile_options(m22-concurrency-tsan PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
target_link_options(m22-concurrency-tsan PRIVATE -fsanitize=thread)
target_link_libraries(m22-concurrency-tsan PRIVATE pthread)
add_test(NAME m22-concurrency-tsan COMMAND m22-concurrency-tsan)
set_tests_properties(m22-concurrency-tsan PROPERTIES TIMEOUT 180)
add_test(NAME m22-host COMMAND "${Python3_EXECUTABLE}" -m pytest -q "${CMAKE_CURRENT_SOURCE_DIR}/test_m22.py" -m "not qemu")
set_tests_properties(m22-host PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../.." TIMEOUT 600)
add_test(NAME m22-bios-uefi-guest COMMAND "${Python3_EXECUTABLE}" -m pytest -q "${CMAKE_CURRENT_SOURCE_DIR}/test_m22.py" -m qemu)
set_tests_properties(m22-bios-uefi-guest PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../.." TIMEOUT 2400)
