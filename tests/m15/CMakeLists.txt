cmake_minimum_required(VERSION 3.22)
project(LamixM15 LANGUAGES C)
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED YES)
set(CMAKE_C_EXTENSIONS NO)
add_compile_options(-Wall -Wextra -Werror -Wpedantic -fsanitize=address,undefined -fno-omit-frame-pointer)
add_link_options(-fsanitize=address,undefined)
include_directories("${CMAKE_CURRENT_SOURCE_DIR}/../../kernel")
enable_testing()
file(MAKE_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../out/m15")
add_executable(m15-native native.c "${CMAKE_CURRENT_SOURCE_DIR}/../../kernel/input.c")
add_test(NAME m15-native-sanitize COMMAND m15-native)
set_tests_properties(m15-native-sanitize PROPERTIES TIMEOUT 180 ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0;UBSAN_OPTIONS=halt_on_error=1:print_stacktrace=1")
add_test(NAME m15-host COMMAND "${Python3_EXECUTABLE}" -m pytest -q "${CMAKE_CURRENT_SOURCE_DIR}/test_m15.py" -m "not qemu" "--junitxml=${CMAKE_CURRENT_SOURCE_DIR}/../../out/m15/host.xml")
set_tests_properties(m15-host PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../.." TIMEOUT 300)
add_test(NAME m15-bios-uefi-guest COMMAND "${Python3_EXECUTABLE}" -m pytest -q "${CMAKE_CURRENT_SOURCE_DIR}/test_m15.py" -m qemu "--junitxml=${CMAKE_CURRENT_SOURCE_DIR}/../../out/m15/guest.xml")
set_tests_properties(m15-bios-uefi-guest PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../.." TIMEOUT 1800 ENVIRONMENT "LAMIX_STAGE2_BUILD=${CMAKE_CURRENT_SOURCE_DIR}/../../build/m15-bios;LAMIX_UEFI_BUILD=${CMAKE_CURRENT_SOURCE_DIR}/../../build/m15-uefi")
