cmake_minimum_required(VERSION 3.22)
project(LamixM23 C)
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED YES)
set(CMAKE_C_EXTENSIONS NO)
add_compile_options(-Wall -Wextra -Werror -Wpedantic)
include_directories("${CMAKE_CURRENT_SOURCE_DIR}/../../kernel")
enable_testing()
get_filename_component(LAMIX_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
file(MAKE_DIRECTORY "${LAMIX_ROOT}/out/m23")
add_executable(m23-native native.c ../../kernel/elf_loader.c ../../kernel/initramfs.c)
target_compile_options(m23-native PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
target_link_options(m23-native PRIVATE -fsanitize=address,undefined)
add_test(NAME m23-native-sanitize COMMAND m23-native)
set_tests_properties(m23-native-sanitize PROPERTIES TIMEOUT 180 ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0;UBSAN_OPTIONS=halt_on_error=1:print_stacktrace=1")
add_test(NAME m23-host COMMAND "${Python3_EXECUTABLE}" -m pytest -q "${CMAKE_CURRENT_SOURCE_DIR}/test_m23.py" -m "not qemu")
set_tests_properties(m23-host PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../.." TIMEOUT 900)
add_test(NAME m23-bios-uefi-guest COMMAND "${Python3_EXECUTABLE}" -m pytest -q "${CMAKE_CURRENT_SOURCE_DIR}/test_m23.py" -m qemu)
set_tests_properties(m23-bios-uefi-guest PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../.." TIMEOUT 3000)
