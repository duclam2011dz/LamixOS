cmake_minimum_required(VERSION 3.22)
project(LamixM24 LANGUAGES C)
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED YES)
set(CMAKE_C_EXTENSIONS NO)
add_compile_options(-Wall -Wextra -Werror -Wpedantic)
include_directories("${CMAKE_CURRENT_SOURCE_DIR}/../../kernel")
enable_testing()
get_filename_component(LAMIX_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
file(MAKE_DIRECTORY "${LAMIX_ROOT}/out/m24")
add_executable(m24-native native.c "${CMAKE_CURRENT_SOURCE_DIR}/../../kernel/scheduler.c" "${CMAKE_CURRENT_SOURCE_DIR}/../../kernel/acpi.c" "${CMAKE_CURRENT_SOURCE_DIR}/../../kernel/process.c" "${CMAKE_CURRENT_SOURCE_DIR}/../../kernel/vmm.c")
target_compile_options(m24-native PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
target_link_options(m24-native PRIVATE -fsanitize=address,undefined)
add_test(NAME m24-native-smp-sanitize COMMAND m24-native)
add_test(NAME m24-host COMMAND "${Python3_EXECUTABLE}" -m pytest -q "${CMAKE_CURRENT_SOURCE_DIR}/test_m24.py" -m "not qemu")
set_tests_properties(m24-host PROPERTIES WORKING_DIRECTORY "${LAMIX_ROOT}" TIMEOUT 1200)
add_test(NAME m24-smp-bios-uefi COMMAND "${Python3_EXECUTABLE}" -m pytest -q "${CMAKE_CURRENT_SOURCE_DIR}/test_m24.py" -m qemu)
set_tests_properties(m24-smp-bios-uefi PROPERTIES WORKING_DIRECTORY "${LAMIX_ROOT}" TIMEOUT 7200)
find_package(Threads REQUIRED)
add_executable(m24-concurrency-tsan concurrency.c "${CMAKE_CURRENT_SOURCE_DIR}/../../kernel/scheduler.c")
target_compile_options(m24-concurrency-tsan PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
target_link_options(m24-concurrency-tsan PRIVATE -fsanitize=thread)
target_link_libraries(m24-concurrency-tsan PRIVATE Threads::Threads)
add_test(NAME m24-concurrency-tsan COMMAND m24-concurrency-tsan)
set_tests_properties(m24-concurrency-tsan PROPERTIES TIMEOUT 180 ENVIRONMENT "TSAN_OPTIONS=halt_on_error=1")
set_tests_properties(m24-native-smp-sanitize PROPERTIES TIMEOUT 180 ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0;UBSAN_OPTIONS=halt_on_error=1:print_stacktrace=1")
